Below is the complete script utilized to convert raw data from 2017 and 2018 to a shared wide data format. This script was developed by using Google Gemini AI. In addition to Tidyverse functions, I also used the “openxlsx” package to read data directly from Excel workbooks. Key features of the script are:

  • Dynamically anchors on word “Date” since that appears in the top-left cell of every sheet. Every time “Date” appears marks the start of new sheet. Returns message with exact tally of calculated sheets.

  • Metadata from rows 1-11 of the raw data are assigned to columns in final table

  • Creates a running list of data from all the separate sheets in its working memory.

  • Reads species count data and pivots this to a long table that contains organism name, quadrat and point information, and total count.

  • Uses regular expressions to identify and isolate transect and point numbers, then returns these to columns. Each new observation point along a transect gets a new row.

  • Cleans species names by converting any value with white spaces to underscores.

  • Drops “junk” species columns that are just numeric strings as well as any value with the word “total” in it.

  • Identifies any species not contained in master list and adds these as new species, which are sorted into new columns alphabetically. This makes it easy to run across the output csv and determine if species duplicates exist, then make manual determinations.

  • Converts excel format date strings into readable MM-DD-YYYY format.

# 2. RUN EXTRACTION LOOP
# ----------------------------------------------------------------------------
all_quadrats_list <- list()

for (idx in 1:length(start_columns)) {
  
  current_form_start_col <- start_columns[idx]
  
  if (idx < length(start_columns)) {
    next_form_label_col <- form_label_cols[idx + 1]
    end_col <- next_form_label_col - 1 
  } else {
    header_row_vals <- readWorkbook(wb, sheet = SHEET_NAME, rows = ORGANISM_HEADER_ROW, 
                                    colNames = FALSE, skipEmptyCols = FALSE)
    end_col <- max(which(!is.na(header_row_vals[1, ]) & header_row_vals[1, ] != ""))
  }
  
  matrix_cols <- (current_form_start_col - 1):end_col
  
  # --- EXTRACT METADATA ---
  meta_data <- list()
  for (meta_field in names(METADATA_ROWS)) {
    val <- suppressWarnings(readWorkbook(wb, sheet = SHEET_NAME, 
                                         rows = METADATA_ROWS[[meta_field]], cols = current_form_start_col, 
                                         colNames = FALSE, rowNames = FALSE))
    meta_data[[meta_field]] <- if(length(val) == 0 || is.na(val)) NA else as.character(val)
  }
  meta_df <- as.data.frame(meta_data, stringsAsFactors = FALSE)
  
  # --- EXTRACT SPECIES GRIDS ---
  raw_grid <- suppressWarnings(readWorkbook(wb, sheet = SHEET_NAME, 
                                            rows = ORGANISM_HEADER_ROW:ORGANISM_END_ROW, 
                                            cols = matrix_cols, 
                                            colNames = FALSE, rowNames = FALSE))
  
  if (is.null(raw_grid) || length(raw_grid) == 0 || nrow(raw_grid) <= 1) {
    next 
  }
  
  # Grab raw column headers
  raw_headers <- as.character(raw_grid[1, ])
  raw_headers[1] <- "Organism" # Only rename the FIRST column, keep the quadrat names!
  
  # Clean and isolate ONLY columns that have real names
  valid_col_indices <- which(!is.na(raw_headers) & raw_headers != "" & raw_headers != "NA")
  
  if (length(valid_col_indices) <= 1) {
    next 
  }
  
  counts_df <- raw_grid[-1, valid_col_indices, drop = FALSE]
  colnames(counts_df) <- raw_headers[valid_col_indices]
  
  # Filter out rows where the Organism name is missing
  counts_df <- counts_df %>% filter(!is.na(Organism) & Organism != "")
  
  if (nrow(counts_df) == 0) {
    next
  }
  
  # --- RESHAPE WIDE GRID TO LONG ROWS ---
  long_counts <- counts_df %>%
    pivot_longer(cols = -Organism, names_to = "quadrat_raw", values_to = "count") %>%
    mutate(
      count = ifelse(is.na(count) | count == "", NA, as.numeric(count)),
      clean_organism = str_to_lower(Organism) %>% 
        str_replace_all(" ", "_") %>% 
        str_replace_all("-", "_") %>%
        str_replace_all("spp", "sp"),
      
      # Standardize strings to lowercase dashes to make parsing bulletproof
      normalized_quadrat = str_to_lower(quadrat_raw) %>% str_replace_all(" ", "-"),
      
      # Strict Look-behind Regex matching to keep column extraction separate
      transect_number = str_extract(normalized_quadrat, "(?<=transect-)\\d+"),
      position_on_transect_m = str_extract(normalized_quadrat, "(?<=point-)\\d+\\.\\d+|(?<=point-)\\d+")
    )
  
  # Group duplicates and summarize counts
  long_counts_clean <- long_counts %>%
    filter(!is.na(transect_number) & !is.na(position_on_transect_m)) %>%
    group_by(transect_number, position_on_transect_m, clean_organism) %>%
    summarise(count = sum(count, na.rm = TRUE), .groups = 'drop')
  
  if (nrow(long_counts_clean) == 0) next
  
  # Pivot organisms back out horizontally into row columns
  wide_quadrats <- long_counts_clean %>%
    pivot_wider(names_from = clean_organism, values_from = count)
  
  # Merge metadata onto the rows
  combined_df <- cbind(meta_df, wide_quadrats)
  all_quadrats_list[[length(all_quadrats_list) + 1]] = combined_df
}

find_sheet <- function(kind, p) { 
  up <- str_to_upper(tab_names) 
  hit <- tab_names[str_detect(up, paste0("^", kind)) & str_detect(up, p)] 
  if (length(hit) != 1) {
    stop("Expected exactly 1 match for kind: ", kind, ", pattern: ", p, 
         ". Tabs in file: ", paste(tab_names, collapse = " | "))
  }
  hit 
}

# 2. Parsing Functions

read_transect <- function(sheet) { 
  read_excel(opihi_file_in, sheet = sheet, range = "A4:I11") |> 
    rename(Point = 1, Location_m = 2) |> 
    mutate(across(everything(), as.character)) |>
    pivot_longer(-c(Point, Location_m), names_to = "Group", values_to = "Original") |> 
    filter(!is.na(Original)) |> 
    filter(!str_detect(str_squish(Original), "^[0-9.]+$")) |> 
    mutate(
      Point = as.character(Point),
      Location_m = suppressWarnings(as.numeric(Location_m)),
      Original = as.character(Original), 
      Method = "Transect", 
      Site = str_sub(Group, 1, 2), 
      Points = 1
    ) 
} 

read_quadrat <- function(sheet) { 
  read_excel(opihi_file_in, sheet = sheet, range = "A5:H40", 
             col_types = c("text", rep("guess", 7))) |> 
    rename(Original = 1) |> 
    filter(!str_detect(str_to_upper(Original), "TOTAL|PERCENT|%|CHECKER")) |>
    filter(!is.na(Original)) |> 
    filter(!str_detect(str_squish(Original), "^[0-9.]+$")) |> 
    mutate(across(-Original, as.character)) |> 
    pivot_longer(-Original, names_to = "Group", values_to = "Points") |> 
    mutate(
      Points = suppressWarnings(as.numeric(Points)),
      Points = replace_na(Points, 0)
    ) |> 
    filter(Points > 0) |> 
    mutate(
      Point = NA_character_,
      Method = "Quadrat", 
      Site = str_sub(Group, 1, 2), 
      Location_m = NA_real_
    ) 
}

read_names_transect <- function(sheet) { 
  nm <- read_excel(opihi_file_in, sheet = sheet, range = "C3:I4", col_names = FALSE, col_types = "text") 
  tibble(
    Method = "Transect", 
    Group = as.character(unlist(nm)), 
    Recorder = as.character(unlist(nm)), 
    Checker_1 = NA_character_, 
    Checker_2 = NA_character_
  ) 
} 

read_names_quadrat <- function(sheet) { 
  nm <- read_excel(opihi_file_in, sheet = sheet, range = "B2:H5", col_names = FALSE, col_types = "text") 
  tibble(
    Method = "Quadrat", 
    Group = as.character(unlist(nm)), 
    Recorder = as.character(unlist(nm)), 
    Checker_1 = as.character(unlist(nm)), 
    Checker_2 = as.character(unlist(nm)))

I also asked AI to help me develop a short piece of code to anonymize the names of individuals in relevant columns. It takes the original master file and converts the unique values in any columns specified to anonymized strings. It then produces an output along with a csv key containing all of the codes linked with the actual names. This csv key can be sent directly to Dr. Philippoff.

Show code 
# --- CONFIGURATION ---
INPUT_FILE             <- here("Data","Master_Output_Dataframe.csv")
ANONYMIZED_OUTPUT_FILE <- here("Data","Master_Output_Dataframe_anonymized.csv")
MAPPING_KEY_FILE       <- here("Data","OPIHI_mapping_key.csv")

# List the exact column names you want to mask
COLUMNS_TO_ANONYMIZE   <- c("team_members","team_lead_or_recorder","data_entry_by") 


# --- CORE LOGIC ---
# 1. Load the original CSV
df <- read_csv(INPUT_FILE)

# Initialize an empty tibble to hold the master lookup map
master_mapping <- tibble(
  Column_Source  = character(),
  Original_Value = character(),
  Randomized_ID  = character()
)

# 2. Loop through each column to generate random maps
for (col in COLUMNS_TO_ANONYMIZE) {
  if (!col %in% colnames(df)) {
    warning(paste("Column '", col, "' not found in CSV. Skipping.", sep=""))
    next
  }
  
  # Extract unique, non-NA values from the targeted column
  unique_vals <- df %>% 
    pull(!!sym(col)) %>% 
    na.omit() %>% 
    unique()
  
  # Generate short, 8-character random alphanumeric IDs
  # e.g., "ID_A1B2C3D4"
  random_ids <- replicate(
    length(unique_vals), 
    paste0("ID_", toupper(paste(sample(c(0:9, letters[1:6]), 8, replace = TRUE), collapse = "")))
  )
  
  # Ensure IDs are perfectly unique inside this loop iteration
  while(any(duplicated(random_ids))) {
    random_ids[duplicated(random_ids)] <- replicate(
      sum(duplicated(random_ids)), 
      paste0("ID_", toupper(paste(sample(c(0:9, letters[1:6]), 8, replace = TRUE), collapse = "")))
    )
  }
  
  # Construct temporary map for this specific column
  col_map <- tibble(
    Column_Source  = col,
    Original_Value = as.character(unique_vals),
    Randomized_ID  = random_ids
  )
  
  # Append to master mapping key
  master_mapping <- bind_rows(master_mapping, col_map)
  
  # Replace the original column values in the dataframe with their new IDs
  df <- df %>%
    left_join(col_map %>% select(Original_Value, Randomized_ID), by = setNames("Original_Value", col)) %>%
    mutate(!!sym(col) := ifelse(is.na(Randomized_ID), !!sym(col), Randomized_ID)) %>%
    select(-Randomized_ID)
}

# 3. Export the processed files
write_csv(df, ANONYMIZED_OUTPUT_FILE)
write_csv(master_mapping, MAPPING_KEY_FILE)

cat("Success! Files generated successfully in your working directory.")

# 1. Load the data
OPIHI_data <- read_csv(here("Data", "Master_Output_Dataframe_anonymized.csv"))

# 2. Fix the character type and fill the years
OPIHI_data <- OPIHI_data |>
  mutate(
    # Force the text "NA" into a true R missing value (NA)
    year = as.numeric(year),
    
    # Extract the 4-digit year from the date string
    extracted_year = as.numeric(str_extract(date, "\\d{4}")),
    
    # Fill in the true NAs with the extracted years
    year = coalesce(year, extracted_year)
  ) |>
  # Remove the temporary column
  select(-extracted_year)
