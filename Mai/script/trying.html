  up  <- str_to_upper(tab_names)
  hit <- tab_names[str_detect(up, paste0("^", kind)) & str_detect(up, p)]

# Transect sheets: header on row 4, 7 points on rows 5-11, columns A:I.
# Everything right of column I is compiled/summary data, so it is skipped.
read_transect <- function(sheet) {
  read_excel(opihi_file_in, sheet = sheet, range = "A4:I11") |>
    rename(Point = 1, Location_m = 2) |>
    pivot_longer(-c(Point, Location_m), names_to = "Group", values_to = "Original") |>
    filter(!is.na(Original)) |>
    mutate(Original = as.character(Original),
           Method = "Transect", Site = str_sub(Group, 1, 2), Points = 1)
}

# Quadrat sheets: header on row 5, organisms start row 6, columns A:H.
# Stop at the "TOTAL Points" row; the TOTALS column (I) and % columns (J:K) are never read.
read_quadrat <- function(sheet) {
  df <- read_excel(opihi_file_in, sheet = sheet, range = "A5:H40") |> rename(Original = 1)
  stop_at <- which(str_detect(df$Original, "TOTAL Points"))[1]
  df[seq_len(stop_at - 1), ] |>
    filter(!is.na(Original)) |>                      # drop empty template rows
    pivot_longer(-Original, names_to = "Group", values_to = "Points") |>
    mutate(Points = replace_na(as.numeric(Points), 0))|>
    filter(Points > 0) |>                            # blank = 0 points
    mutate(Method = "Quadrat", Site = str_sub(Group, 1, 2), Location_m = NA_real_)
}

# Each survey is a block of columns that starts where row 1 says "Date".
# In every block:  rows 1-11 = info (date, location, lat, long, school) in the column after the label
#row 13 = quadrat headers ("transect-1 quadrat-1 at point 1.00")
#rows 14-60 = organism names (first column of block) and counts
#rows 61+  = TOTALS section (skipped: not raw data)
starts <- which(m[1, ] == "Date")
ends   <- c(starts[-1] - 1, ncol(m))
 
# helpers
to_date <- function(x) { # Excel dates arrive as serial numbers when read as text
  d <- suppressWarnings(as.numeric(x))
  if (!is.na(d)) as.Date(d, origin = "1899-12-30") else as.Date(substr(x, 1, 10))
}
 
read_block <- function(c0, c1) {
  info_col <- c0 + 1
  cols <- (c0 + 1):c1
  cols <- cols[!is.na(m[13, cols])]                                   # only columns with a quadrat header
  rows <- 14:60
  rows <- rows[!is.na(m[rows, c0]) & m[rows, c0] != ""]               # only rows with an organism name
 
  body <- as.data.frame(m[rows, cols, drop = FALSE], stringsAsFactors = FALSE)
  names(body) <- paste0("c", seq_along(cols))
  body$original_labels <- m[rows, c0]
  header <- m[13, cols]
 
  body |>
    pivot_longer(-original_labels, names_to = "col", 
                 values_to = "points") |>
    mutate(points = suppressWarnings(as.numeric(points))) |>
    filter(!is.na(points), points > 0) |>   # blank = not observed
   mutate(header = header[as.integer(str_remove(col, "^c"))],
           date      = to_date(m[1, info_col]),
           site      = m[8, info_col],
           latitude  = as.numeric(m[9, info_col]),
           longitude = as.numeric(m[10, info_col]),
           school    = m[11, info_col],
           number_of_transect_lines = as.numeric(m[4, info_col]),
           length_of_transect_lines = as.numeric(m[5, info_col]),
           point_spacing            = m[6, info_col],
           number_of_quadrats_per_transect = as.numeric(m[7, info_col]),
           transect  = as.integer(str_match(header, "transect-(\\d+)")[, 2]),
           quadrat   = as.integer(str_match(header, "quadrat-(\\d+)")[, 2]),
           transect_points = as.numeric(str_match(header, "point ([0-9.]+)")[, 2]))
}

  cols <- cols[!is.na(m[13, cols])]  # columns with a quadrat header (row 13)
  totals_row <- which(m[, c0] == "TOTALS:")[1] # TOTALS section = not raw data
  rows <- 14:(totals_row - 2) # organism rows end 2 rows above TOTALS
  rows <- rows[!is.na(m[rows, c0]) & m[rows, c0] != ""]
  if (length(cols) == 0 || length(rows) == 0) return(NULL) # empty survey (e.g. Kahana Bay)
 
  body <- as.data.frame(m[rows, cols, drop = FALSE], stringsAsFactors = FALSE)
  names(body) <- paste0("c", seq_along(cols))
  body$original_labels <- m[rows, c0]
  header <- m[13, cols]

  body |>
    pivot_longer(-original_labels, names_to = "col", 
                 values_to = "points") |>
    mutate(points = suppressWarnings(as.numeric(points))) |>
    filter(!is.na(points), points > 0) |>   # blank = not observed
    mutate(header = header[as.integer(str_remove(col, "^c"))],
           date      = to_date(m[1, info]),
           site      = m[8, info],
           latitude  = as.numeric(m[9, info]),
           longitude = as.numeric(m[10, info]),
           school    = str_squish(m[11, info]),
           number_of_transect_lines        = as.numeric(m[4, info]),
           length_of_transect_lines        = as.numeric(m[5, info]),
           point_spacing                   = m[6, info],
           number_of_quadrats_per_transect = as.numeric(m[7, info]),
           transect = as.integer(str_match(header, "transect-(\\d+)")[, 2]),
           quadrat = as.integer(str_match(header, "quadrat-(\\d+)")[, 2]),
           transect_points = as.numeric
           (str_match(header,"point ([0-9.]+)")[, 2]))
}
