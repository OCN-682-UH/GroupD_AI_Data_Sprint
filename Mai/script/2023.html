find_sheet <- function(kind, p) {
  up  <- str_to_upper(tab_names) 
  hit <- tab_names[str_detect(up, paste0("^", kind)) & str_detect(up, p)]
  if (length(hit) != 1)
    stop("Expected exactly 1 '", kind, " ... ", p, "' tab, found ", length(hit),
         ". Tabs in file: ", paste(tab_names, collapse = " | "))
  hit
}

# Raw Data
# Transect sheets: header on row 4, 7 points on rows 5-11, columns A:I.
# Everything right of column I is compiled/summary data, so it is skipped.
read_transect <- function(sheet) {
  read_excel(opihi_file_in, sheet = sheet, range = "A4:I11") |>
    rename(Point = 1, Location_m = 2) |>
    pivot_longer(-c(Point, Location_m), names_to = "Group", values_to = "Original") |>
    filter(!is.na(Original)) |>
    mutate(Original = as.character(Original),
           Method = "Transect", Site = str_sub(Group, 1, 2), Points = 1)
}

# Quadrat sheets: header on row 5, organisms start row 6, columns A:H.
# Stop at the "TOTAL Points" row; the TOTALS column (I) and % columns (J:K) are never read.
read_quadrat <- function(sheet) {
  df <- read_excel(opihi_file_in, sheet = sheet, range = "A5:H40") |> rename(Original = 1)
  stop_at <- which(str_detect(df$Original, "TOTAL Points"))[1]
  df[seq_len(stop_at - 1), ] |>
    filter(!is.na(Original)) |>                      # drop empty template rows
    pivot_longer(-Original, names_to = "Group", values_to = "Points") |>
    mutate(Points = replace_na(as.numeric(Points), 0))|>
    filter(Points > 0) |>                            # blank = 0 points
    mutate(Method = "Quadrat", Site = str_sub(Group, 1, 2), Location_m = NA_real_)
}

# Transect sheets: NAME in row 3 (C:I), group codes in row 4.
# Quadrat sheets:  NAME / CHECKER #1 / CHECKER #2 in rows 2-4 (B:H), group codes in row 5.
#function to read transect names
read_names_transect <- function(sheet) {
  nm <- read_excel(opihi_file_in, sheet = sheet, range = "C3:I4", col_names = FALSE, col_types = "text")
  tibble(Method = "Transect",
         Group = as.character(unlist(nm[2, ])),
         Recorder = as.character(unlist(nm[1, ])),
         Checker_1 = NA_character_, Checker_2 = NA_character_)
}

#function to read the quadrat names 
read_names_quadrat <- function(sheet) {
  nm <- read_excel(opihi_file_in, sheet = sheet, range = "B2:H5", col_names = FALSE, col_types = "text")
  tibble(Method = "Quadrat",
         Group = as.character(unlist(nm[4, ])),
         Recorder = as.character(unlist(nm[1, ])),
         Checker_1 = as.character(unlist(nm[2, ])),
         Checker_2 = as.character(unlist(nm[3, ])))
}

clean_label <- function(x) {
  key <- str_to_lower(str_squish(x))
  out <- unname(clean_map[key])
  ifelse(is.na(out), str_squish(x), out)# unmapped labels are just trimmed
}
